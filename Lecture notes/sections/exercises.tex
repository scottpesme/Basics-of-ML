\section{Exercises}
\label{sec:exercises}

This section collects additional exercises.


\begin{exercise}[$\star$]
Let $A \in \R^{d \times d}$ and let $f(w) = w^\top A w$. Prove that $f(w) = \frac{1}{2} w^\top (A + A^\top) w$.
\end{exercise}

\begin{exercise}[$\star$]
All convex functions have a global minimiser. True or False?
\end{exercise}

\begin{exercise}[$\star$]
Let $f : \R^2 \to \R$ be defined as $f(w_1, w_2) = w_1 w_2$. 
\begin{enumerate}
    \item Draw the level lines of $f$ in the $\R^2$ plane. Using these level lines, prove that $f$ is not convex.
    \item Prove that $f$ can be written $f(w) = w^\top A w$ where $A \in \R^{2\times 2}$. 
    \item Is $A$ unique? If not, provide the corresponding set of matrices $A$.
    \item This time using $A$, prove that $f$ is not convex.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]
Let $A \in \R^{d \times d}$ be symmetric with eigenvalues $\lambda_1 \geq \dots \geq
\lambda_d$. Show that for every $w \in \R^d$,
\[
\lambda_d \Vert w \Vert^2 \ \leq \ w^\top A w \ \leq \ \lambda_1 \Vert w \Vert^2 ,
\]
and that both bounds are attained. Deduce that $A \succeq 0$ if and only if all its
eigenvalues are non-negative.
\end{exercise}

\begin{exercise}[$\star\star$]
Let $A \in \R^{d \times d}$ be symmetric with $A \succeq 0$, let $b \in \R^d$, and
consider $f(w) = \frac{1}{2} w^\top A w - \langle b, w \rangle$.
\begin{enumerate}
    \item Show that $f$ is bounded below if and only if $b \in \mathrm{span}(A)$ (the span generated by the columns of the matrix $A$), and that in
    that case its set of minimisers is exactly $\{ w : Aw = b \}$, an affine space of
    direction $\ker(A)$.
    \item Exhibit an $A$ and a $b$ in $\R^2$ for which $f$ is convex but has no
    minimiser. 
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]
Run gradient descent $w_{t+1} = w_t - \gamma \nabla f(w_t)$ on the quadratic $f(w) =
\frac{1}{2}(w - w^\star)^\top A(w - w^\star)$, with $A \succeq 0$ symmetric.
\begin{enumerate}
    \item Show that $w_t - w^\star = (I - \gamma A)^t (w_0 - w^\star)$.
    \item Assume $A \succ 0$ and let $\lambda_1 \geq \dots \geq \lambda_d > 0$ be its
    eigenvalues. Show that $w_t \to w^\star$ for every $w_0$ if and only if $0 <
    \gamma < 2/\lambda_1$. This is the threshold announced in
    \Cref{rem:step-size}, which we had admitted.
    \item Still for $A \succ 0$, the speed of convergence is governed by $\rho(\gamma)
    = \max_j \vert 1 - \gamma \lambda_j \vert$. Show that $\rho$ is minimised at
    $\gamma^\star = \frac{2}{\lambda_1 + \lambda_d}$, and that then $\rho = \frac{\kappa -
    1}{\kappa + 1}$ where $\kappa = \lambda_1 / \lambda_d$ is the condition number of
    $A$. What does a large $\kappa$ mean geometrically for the level sets of $f$?
    \item Now drop the assumption $A \succ 0$. What happens to the component of $w_0 -
    w^\star$ lying in $\ker(A)$? 
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]
Fit an intercept explicitly, i.e.\ minimise $L(w, b) = \frac{1}{2n}\sum_{i=1}^n
\big( \langle w, x_i \rangle + b - y_i \big)^2$ over $(w, b) \in \R^d \times \R$.
\begin{enumerate}
    \item Show that at any minimiser the residuals sum to zero, and that $b^\star =
    \bar y - \langle w^\star, \bar x \rangle$, where $\bar x = \frac1n \sum_i x_i$ and
    $\bar y = \frac1n \sum_i y_i$.
    \item Substituting $b^\star$ back, show that $w^\star$ minimises $\frac{1}{2n}
    \sum_i \big( \langle w, x_i - \bar x \rangle - (y_i - \bar y) \big)^2$. In other
    words: fitting an intercept is exactly the same as centring the data and
    forgetting about the intercept.
    \item Deduce the closed form in dimension $d = 1$: $w^\star = \frac{\sum_i (x_i -
    \bar x)(y_i - \bar y)}{\sum_i (x_i - \bar x)^2}$, provided the $x_i$ are not all
    equal.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]
Append one extra feature to the data, i.e.\ replace $X \in \R^{n \times d}$ by $X' =
(X \ \vert \ v) \in \R^{n \times (d+1)}$ for some $v \in \R^n$. Show that the
minimal training loss can only decrease, whatever $v$ is, even if $v$ is pure
noise, independent of $y$. Is the same true of the error on unseen data? 
\end{exercise}

\begin{exercise}[$\star\star$]
Let $0 < \lambda_1 < \lambda_2$ and write $w_1, w_2$ for the corresponding ridge
solutions. By writing the optimality of each one against the other, show that
\[
\Vert w_2 \Vert \leq \Vert w_1 \Vert
\qquad \text{and} \qquad
L(w_1) \leq L(w_2) .
\]
So increasing $\lambda$ shrinks the weights and worsens the training loss,
monotonically. Why does this not mean that $\lambda$ should be taken as small as
possible?
\end{exercise}
